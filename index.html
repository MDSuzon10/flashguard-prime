async function askAI() { const input = document.getElementById('user-input'); const chat = document.getElementById('chat'); const cmd = input.value; if(!cmd) return; chat.innerHTML += `
You: ${cmd}
`; input.value = ""; // জেমিনি এআই-কে নির্দেশ পাঠানো const prompt = `You are a smart security assistant. If user wants to turn on light, reply ONLY: {"action":"on"}. If off, reply ONLY: {"action":"off"}. Otherwise reply in Bengali. User command: ${cmd}`; try { // এন্ডপয়েন্ট v1beta থেকে v1 এ পরিবর্তন করে দেখা যেতে পারে const response = await fetch(`https://generativelanguage.googleapis.com/v1/models/gemini-1.5-flash:generateContent?key=${GEMINI_API_KEY}`, { method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify({ "contents": [{ "parts": [{"text": prompt}] }] }) }); const data = await response.json(); // যদি গুগল থেকে কোনো এরর মেসেজ আসে তা কনসোলে দেখাবে if (data.error) { console.error("Google API Error:", data.error.message); chat.innerHTML += `
Error: ${data.error.message}
`; return; } const aiRes = data.candidates[0].content.parts[0].text; if(aiRes.includes('"action":"on"')) { db.ref('system/police_light').set(1); chat.innerHTML += `
AI: জি স্যার, লাইট জ্বালিয়ে দিয়েছি।
`; } else if(aiRes.includes('"action":"off"')) { db.ref('system/police_light').set(0); chat.innerHTML += `
AI: ঠিক আছে, লাইট বন্ধ করা হয়েছে।
`; } else { chat.innerHTML += `
AI: ${aiRes}
`; } } catch (e) { console.error("Connection Error:", e); chat.innerHTML += `
Error: এআই সংযোগে সমস্যা হচ্ছে। কনসোল চেক করুন।
`; } chat.scrollTop = chat.scrollHeight; }
